pipeline {
    agent any

    // En este fichero voy a explicar cada pregunta que se plantea en el reto 1

    stages {

        // El reto 1 indica que se debe obtener el código fuente desde el repositorio.
        stage('Get Code') {
            steps {
                echo '########## GET CODE ##########'
                git 'https://github.com/pablotose/helloworld-cp1-unir.git'
            }
        }

        // En el reto 1, para las pruebas unitarias se especifica que no se cuenta con ningún baremo,
        // por lo que siempre se mostrará en verde.
        // Además, las pruebas unitarias solo se pueden ejecutar una sola vez en el mismo pipeline.
        // Por eso aquí hacemos "coverage run + pytest + junitxml" TODO en una sola ejecución.
        stage('Unit') {
            steps {
                echo '########## UNIT ##########'
                // Siempre verde, aunque falle (según enunciado)
                catchError(buildResult: 'SUCCESS', stageResult: 'SUCCESS') {
                    bat '''
                        set PYTHONPATH=%WORKSPACE%
                        "C:\\Users\\Pablo\\AppData\\Local\\Programs\\Python\\Python311\\python.exe" -m coverage run --branch --source=app --omit=app\\__init__.py,app\\api.py -m pytest --junitxml=result-unit.xml test\\unit
                    '''
                }
                // Publicamos resultados aunque el xml no exista (para evitar que el pipeline rompa por esto)
                junit allowEmptyResults: true, testResults: 'result-unit.xml'
            }
        }

        // Pruebas de integración (Rest) igual que en el CP1.1.
        // El enunciado indica que esta etapa siempre debe finalizar en verde aunque falle.
        stage('Rest') {
            steps {
                echo '########## REST ##########'
                // Siempre verde, aunque falle
                catchError(buildResult: 'SUCCESS', stageResult: 'SUCCESS') {
                    bat '''
                        set FLASK_APP=app\\api.py

                        start "flask" /B "C:\\Users\\Pablo\\AppData\\Local\\Programs\\Python\\Python311\\python.exe" -m flask run
                        start "wiremock" /B java -jar C:\\Users\\Pablo\\Documents\\Unir\\CP1\\helloworld\\test\\wiremock\\wiremock-standalone-3.13.2.jar --port 9090 --root-dir test\\wiremock

                        ping -n 10 127.0.0.1

                        set PYTHONPATH=%WORKSPACE%
                        "C:\\Users\\Pablo\\AppData\\Local\\Programs\\Python\\Python311\\python.exe" -m pytest --junitxml=result-rest.xml test\\rest
                    '''
                }
                // Publicamos resultados aunque el xml no exista (para evitar que el pipeline rompa por esto)
                junit allowEmptyResults: true, testResults: 'result-rest.xml'
            }
        }

        // Pruebas de análisis de código estático con flake8.
        // El enunciado indica:
        // - Si hay 8 o más hallazgos => UNSTABLE
        // - Si hay 10 o más hallazgos => FAILURE
        // - El pipeline debe continuar siempre.
        stage('Static') {
            steps {
                echo '########## STATIC ##########'
                bat '''
                    "C:\\Users\\Pablo\\AppData\\Local\\Programs\\Python\\Python311\\python.exe" -m flake8 --exit-zero --format=pylint app > flake8.out
                '''
                // Siempre continúa el pipeline, aunque supere umbrales
                catchError(buildResult: 'SUCCESS', stageResult: 'SUCCESS') {
                    recordIssues(
                        tools: [flake8(name: 'Flake8', pattern: 'flake8.out')],
                        qualityGates: [
                            [threshold: 8,  type: 'TOTAL', unstable: true],
                            [threshold: 10, type: 'TOTAL', unstable: false]
                        ]
                    )
                }
            }
        }

        // Pruebas de seguridad (Bandit).
        // El enunciado indica:
        // - Si hay 2 o más hallazgos => UNSTABLE
        // - Si hay 4 o más hallazgos => FAILURE
        // - El pipeline debe continuar siempre.
        stage('Security Test') {
            steps {
                echo '########## SECURITY ##########'
                bat '''
                    "C:\\Users\\Pablo\\AppData\\Local\\Programs\\Python\\Python311\\python.exe" -m bandit --exit-zero -r . -f custom -o bandit.out --msg-template "{abspath}:{line}: [{test_id}] {msg}"
                '''
                // Siempre continúa el pipeline, aunque supere umbrales
                catchError(buildResult: 'SUCCESS', stageResult: 'SUCCESS') {
                    recordIssues(
                        tools: [pyLint(name: 'Bandit', pattern: 'bandit.out')],
                        qualityGates: [
                            [threshold: 2, type: 'TOTAL', unstable: true],
                            [threshold: 4, type: 'TOTAL', unstable: false]
                        ]
                    )
                }
            }
        }

        // Pruebas de rendimiento (JMeter).
        // El enunciado indica que flask debe estar levantado y se ejecuta el test-plan en esta etapa.
        stage('Performance') {
            steps {
                echo '########## PERFORMANCE ##########'
                // No paramos el pipeline si falla (debe continuar)
                catchError(buildResult: 'SUCCESS', stageResult: 'SUCCESS') {
                    bat '''
                        set FLASK_APP=app\\api.py

                        start "flask" /B "C:\\Users\\Pablo\\AppData\\Local\\Programs\\Python\\Python311\\python.exe" -m flask run
                        ping -n 10 127.0.0.1

                        C:\\Users\\Pablo\\Documents\\Unir\\CP1\\apache-jmeter-5.6.3\\bin\\jmeter -n -t test\\jmeter\\flask.jmx -f -l flask.jtl
                    '''
                }
                perfReport sourceDataFiles: 'flask.jtl'
            }
        }

        // Pruebas de cobertura (Coverage).
        // Importante: no se vuelven a ejecutar las unit aquí, solo se genera el XML y se publica.
        // Umbrales del enunciado:
        // - Líneas: 85-95 unstable, >95 verde, <85 rojo
        // - Ramas: 80-90 unstable, >90 verde, <80 rojo
        // El pipeline debe continuar siempre.
        stage('Coverage') {
            steps {
                echo '########## COVERAGE ##########'
                bat '''
                    "C:\\Users\\Pablo\\AppData\\Local\\Programs\\Python\\Python311\\python.exe" -m coverage xml
                '''
                // Siempre continúa el pipeline, aunque no cumpla umbrales
                catchError(buildResult: 'SUCCESS', stageResult: 'SUCCESS') {
                    recordCoverage(
                        qualityGates: [
                            [criticality: 'ERROR', metric: 'LINE',   threshold: 85.0],
                            [metric: 'LINE',       threshold: 95.0],
                            [criticality: 'ERROR', metric: 'BRANCH', threshold: 80.0],
                            [metric: 'BRANCH',     threshold: 90.0]
                        ],
                        tools: [[parser: 'COBERTURA', pattern: 'coverage.xml']]
                    )
                }
            }
        }
    }
}
